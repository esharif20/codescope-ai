```python
from __future__ import annotations

from dataclasses import dataclass, field

import networkx as nx

from app.analysis.models import ResolvedCall, Symbol, UnresolvedCall


@dataclass
class CallGraph:
    """D-12: the queryable result. `.graph` is the raw networkx.DiGraph for callers
    that need something the named helpers below don't anticipate -- always keep it
    accessible, never hide it behind only the helpers."""

    graph: nx.DiGraph
    unresolved: dict[str, list[UnresolvedCall]] = field(default_factory=dict)


def build_call_graph(
    symbols_by_module: dict[str, list[Symbol]],
    resolved_calls: list[ResolvedCall],
    unresolved_calls: list[UnresolvedCall],
) -> CallGraph:
    """D-09/D-10/D-11:
    - One node per Symbol across every module, id = f"{module}:{symbol.qualified_name}"
      (D-09 -- classes included, not just functions/methods). Store the Symbol's
      `kind`, `file_path`, `start_line`, `end_line` as node attributes (plain values,
      not the Symbol object itself, so the graph stays trivially serializable).
    - For each ResolvedCall: if BOTH `caller` and `callee` exist as nodes, add or
      update one edge. If the edge does not exist yet, create it with attribute
      `lines=[call.line]`. If it already exists, APPEND `call.line` to the existing
      `lines` list rather than creating a duplicate edge or a duplicate line entry
      (D-10 -- one DiGraph edge per caller->callee pair; if the identical line
      number is somehow submitted twice, de-duplicate it too -- a sorted list of
      unique ints). If either endpoint is missing from the node set (should not
      happen given Plan 06's design, but never silently invent a node for it),
      skip that call and do not add a partial edge.
    - Group every UnresolvedCall by its `enclosing_symbol` into `CallGraph.unresolved`
      (a plain dict, append -- never deduplicate these, every call site's own entry
      matters).
    """


def callers_of(call_graph: CallGraph, symbol_id: str) -> list[str]:
    """Direct predecessors of `symbol_id`, sorted alphabetically for determinism
    (AGENTS.md rule 3) regardless of edge-insertion order. Empty list if the id has
    no callers or does not exist in the graph (never raises for an unknown id)."""


def callees_of(call_graph: CallGraph, symbol_id: str) -> list[str]:
    """Direct successors of `symbol_id`, sorted alphabetically. Same no-raise
    contract as `callers_of`."""


def reverse_reachable_from(call_graph: CallGraph, symbol_id: str) -> list[str]:
    """Every node that can reach `symbol_id` via one or more directed edges
    (`networkx.ancestors`), sorted alphabetically. This is the primitive Phase 3's
    reverse-impact tracing (changed symbol -> affected routes) will call directly."""
```

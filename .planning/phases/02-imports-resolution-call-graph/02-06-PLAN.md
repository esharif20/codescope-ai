```python
from __future__ import annotations

from app.analysis.calls import RawCall, RawClassBases
from app.analysis.models import ImportBinding, ResolvedCall, Symbol, UnresolvedCall


def build_class_hierarchy(
    raw_bases_by_module: dict[str, list[RawClassBases]],
    bindings_by_module: dict[str, list[ImportBinding]],
) -> dict[str, list[str]]:
    """D-01: map every class's id (`module_path:qualified_name`) to the ordered list
    of its DIRECT base classes' ids, but only bases that resolve to a class DEFINED
    IN THIS REPO (external/unresolved bases are simply omitted from the list -- not
    an error, since D-01 only needs repo-defined bases).

    For each RawClassBases entry (module M, class C, raw base name B):
    - If B has no "." : first check whether M itself defines a top-level class named
      B (same-module base, no import needed). If not, look up B in
      `bindings_by_module[M]` by `imported_name`; if that binding's status is
      RESOLVED, the base id is `f"{binding.source_module}:{binding.source_name}"`.
    - If B contains "." (e.g. "mod.Base2"): split on the FIRST "." into
      (module_alias, rest). Look up `module_alias` in `bindings_by_module[M]` by
      `imported_name`; if that binding's status is RESOLVED and its `source_name is
      None` (i.e. it bound a whole module, not a single name -- e.g. `import mod`),
      the base id is `f"{binding.source_module}:{rest}"`.
    - Any other case (external, unresolved, or no matching binding at all): omit
      this base from the result entirely.
    Returns one key per class that has at least one in-repo-resolved base; classes
    with zero in-repo bases simply have no key (callers must use `.get(id, [])`).
    """


def resolve_file_calls(
    raw_calls: list[RawCall],
    importing_module: str,
    bindings: list[ImportBinding],
    symbol_table: dict[str, list[Symbol]],
    class_hierarchy: dict[str, list[str]],
) -> tuple[list[ResolvedCall], list[UnresolvedCall]]:
    """D-01..D-04: resolve one file's raw calls. `symbol_table` maps every in-repo
    module_path to its full Phase 1 Symbol list (used both for same-module lookups
    and for resolving targets in OTHER modules via `bindings`).

    For each RawCall:

    - receiver_kind == "self" or "cls": let `name` = the text after the dot. Find
      the Symbol in `symbol_table[importing_module]` whose `qualified_name ==
      raw_call.enclosing_symbol`; its `parent_qualified_name` is the directly
      enclosing class's qualified_name -- if None, emit UNRESOLVED (self/cls used
      outside a method body is a defensible gap, not a crash). Build the search
      order: `[f"{importing_module}:{parent_qualified_name}"]`, then a left-to-right
      preorder walk of `class_hierarchy` from that id (visit each base id in list
      order, recursing into ITS bases before moving to the next sibling, never
      visiting the same id twice). For each candidate class id `"{mod}:{cls}"` in
      that order, look for a Symbol in `symbol_table[mod]` with
      `qualified_name == f"{cls}.{name}"`. First match -> one ResolvedCall. No match
      anywhere in the order -> one UnresolvedCall (status=UNRESOLVED,
      expression=raw_call.expression).

    - receiver_kind == "bare": let `name` = raw_call.expression (no dot). Candidates,
      in this priority order: (1) a top-level Symbol (parent_qualified_name is None)
      named `name` in `symbol_table[importing_module]` (same-module function or
      class); (2) if not found in (1), every ImportBinding in `bindings` whose
      `imported_name == name` AND whose `status == RESOLVED` (there may be more than
      one if, hypothetically, duplicate bindings existed -- normally at most one).
      For each such binding, look up a top-level Symbol named `binding.source_name`
      (or, if `source_name is None` -- a plain `import x` -- the module itself has
      no single symbol to point at; skip it, this phase does not resolve calls to
      bare module names) in `symbol_table[binding.source_module]`. Collect every
      distinct resolved target symbol id found this way (same-module match from (1)
      counts as one candidate too, id = `f"{importing_module}:{name}"`). If exactly
      one candidate -> one ResolvedCall. If two or more DISTINCT candidates -> one
      UnresolvedCall with `status=UNCERTAIN` and `candidates` = the sorted list of
      every candidate id (D-03 -- never multiple edges, never a guess). If zero
      candidates -> one UnresolvedCall with `status=UNRESOLVED` (covers builtins,
      parameters, locals, dynamic dispatch targets -- D-02/D-14).

    - receiver_kind == "attribute": let `object_name, attr_name` = split
      raw_call.expression on the FIRST "." (e.g. `"obj.baz"` -> `"obj"`, `"baz"`;
      `"a.b.c"` -> `"a"`, `"b.c"`). Look up `object_name` in `bindings` by
      `imported_name`. If found AND `status == RESOLVED` AND `source_name is None`
      (a whole-module import, e.g. `import services.auth as auth_mod` then
      `auth_mod.get_current_user()`): look up a top-level Symbol named `attr_name`
      (use only the portion after the first "." as a single lookup key -- do not
      resolve further dotted chains beyond one level for this phase) in
      `symbol_table[binding.source_module]`; found -> ResolvedCall, else ->
      UnresolvedCall(status=UNRESOLVED). In EVERY other case (no matching binding,
      binding not a whole-module import, binding external or unresolved, or
      `object_name` not found in `bindings` at all -- meaning it is a local
      variable, parameter, `self`/`cls` typo, or anything else not statically
      knowable) -> UnresolvedCall(status=UNRESOLVED) (D-02's core case, and D-04's
      "no type/variable tracking" guarantee: an instance assigned from a
      constructor call, e.g. `auth = AuthService(); auth.login()`, is NEVER found
      in `bindings` by the name `"auth"`, so it always falls through to here).

    All IDs are `module_path:qualified_name`; `ResolvedCall.caller` and
    `UnresolvedCall.enclosing_symbol` are always
    `f"{importing_module}:{raw_call.enclosing_symbol}"`.
    """
```
